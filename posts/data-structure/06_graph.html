图 (Graph) —— 数据结构核心详解与Go语言实现

图的定义

图(Graph)由两个集合构成，分别是顶点集合V和边集合E，记为 G = (V, E)。
顶点集合 V = {A, B, C, D, E}，边集合 E = {(A,B), (A,C), (A,D), (B,C), (B,E), (C,D), (C,E)}。
顶点集合不能为空，边集合可以为空。线性表可以是空表，树可以是空树，但图至少要有一个顶点。

图的术语

  • 无向图：每条边都是无方向的。
  • 有向图：每条边都是有方向的。
  • 完全图：任意两个点之间都有一条边相连。
  • 稀疏图：有很少边或弧的图（e < n log n）。
  • 稠密图：有较多边或弧的图。
  • 网：边/弧带权的图。
  • 邻接：有边/弧相连的两个顶点之间的关系。
  • 关联(依附)：边/弧与顶点之间的关系。
  • 顶点的度：与该顶点相关联的边的数目，记为 TD(V)。在有向图中，顶点的度等于入度与出度之和。顶点v的入度是以v为终点的有向边的条数，记作 ID(V)；出度是以v为始点的有向边的条数，记作 OD(V)。
  • 路径：接续的边构成的顶点序列。
  • 路径长度：路径上边或弧的数目/权值之和。
  • 回路(环)：第一个顶点和最后一个顶点相同的路径。
  • 简单路径：除路径起点和终点可以相同外，其余顶点均不相同的路径。
  • 简单回路(简单环)：除路径起点和终点相同外，其余顶点均不相同的路径。
  • 权与网：图中边或弧所具有的相关数称为权，带权的图称为网。
  • 连通图 (强连通图)：在无(有)向图 G = (V, {E}) 中，若对任何两个顶点 v、u 都存在从 v 到 u 的路径，则称 G 是连通图（强连通图）。
  • 连通分量（强连通分量）：无向图G的极大连通子图称为G的连通分量；有向图G的极大强连通子图称为G的强连通分量。
  • 生成树：包含无向图G所有顶点的极小连通子图。
  • 生成森林：对非连通图，由各个连通分量的生成树的集合。

图的抽象数据类型 (ADT)

ADT Graph {
    数据对象V：具有相同特性的数据元素的集合，称为顶点集。
    数据关系R：

图的遍历

深度优先遍历 (DFS)

递归或借助栈实现，类似树的先序遍历。

广度优先遍历 (BFS)

借助队列实现，类似树的层序遍历。

图的应用

最小生成树 — Prim（普里姆）算法

从某一顶点出发，每次选择代价最小的边扩展至顶点集，适用于稠密图。

最小生成树 — Kruskal（克鲁斯卡尔）算法

按权值递增顺序选择边，利用并查集避免环路，适用于稀疏图。

最短路径 — Dijkstra（迪杰斯特拉）算法

单源最短路径，贪心策略，要求权值非负。

最短路径 — Floyd（弗洛伊德）算法

多源最短路径，动态规划，可处理负权边（无负环）。

拓扑排序 (Topological Sort)

对有向无环图(DAG)顶点线性排序，常用Kahn算法（基于入度）或DFS。

AOE网和关键路径 (Critical Path)

AOE网(Activity On Edge)中，关键路径是决定工程完工时间的最长路径，通过事件最早/最晚时间计算关键活动。

完整Go语言实现：图的基础算法集

以下代码实现了无向/有向图的构建、DFS/BFS、Prim/Kruskal最小生成树、Dijkstra/Floyd最短路径、拓扑排序及AOE关键路径，并包含完整测试用例。

package main

import (
	"container/heap"
	"fmt"
	"math"
)

// ---------------------------- 图的基础定义 ----------------------------

// Edge 表示图中的边
type Edge struct {
	to     int // 目标顶点
	weight int // 边的权重（无权图可设为1）
}

// Graph 表示图，使用邻接表存储
type Graph struct {
	adjacency map[int][]Edge // 邻接表：顶点 -> 边列表
	vertices  map[int]bool   // 存储所有顶点
}

// NewGraph 创建一个新的图
func NewGraph() *Graph {
	return &Graph{
		adjacency: make(map[int][]Edge),
		vertices:  make(map[int]bool),
	}
}

// AddVertex 添加顶点
func (g *Graph) AddVertex(v int) {
	if !g.vertices[v] {
		g.vertices[v] = true
		g.adjacency[v] = []Edge{}
	}
}

// AddEdge 添加有向边
func (g *Graph) AddEdge(from, to, weight int) {
	g.AddVertex(from)
	g.AddVertex(to)
	g.adjacency[from] = append(g.adjacency[from], Edge{to: to, weight: weight})
}

// AddUndirectedEdge 添加无向边
func (g *Graph) AddUndirectedEdge(from, to, weight int) {
	g.AddEdge(from, to, weight)
	g.AddEdge(to, from, weight)
}

// ---------------------------- 1. 深度优先遍历（DFS） ----------------------------

// DFS 深度优先遍历（递归实现）
func (g *Graph) DFS(start int) []int {
	visited := make(map[int]bool)
	result := []int{}
	var dfs func(v int)
	dfs = func(v int) {
		if visited[v] {
			return
		}
		visited[v] = true
		result = append(result, v)
		for _, edge := range g.adjacency[v] {
			dfs(edge.to)
		}
	}
	dfs(start)
	return result
}

// DFSIterative 深度优先遍历（非递归实现，使用栈）
func (g *Graph) DFSIterative(start int) []int {
	visited := make(map[int]bool)
	stack := []int{start}
	result := []int{}
	for len(stack) > 0 {
		v := stack[len(stack)-1]
		stack = stack[:len(stack)-1]
		if !visited[v] {
			visited[v] = true
			result = append(result, v)
			for i := len(g.adjacency[v]) - 1; i >= 0; i-- {
				edge := g.adjacency[v][i]
				if !visited[edge.to] {
					stack = append(stack, edge.to)
				}
			}
		}
	}
	return result
}

// ---------------------------- 2. 广度优先遍历（BFS） ----------------------------

// BFS 广度优先遍历（使用队列）
func (g *Graph) BFS(start int) []int {
	visited := make(map[int]bool)
	queue := []int{start}
	visited[start] = true
	result := []int{}
	for len(queue) > 0 {
		v := queue[0]
		queue = queue[1:]
		result = append(result, v)
		for _, edge := range g.adjacency[v] {
			if !visited[edge.to] {
				visited[edge.to] = true
				queue = append(queue, edge.to)
			}
		}
	}
	return result
}

// ---------------------------- 3. 最小生成树：Prim算法 ----------------------------

type PrimItem struct {
	key    int
	vertex int
}
type PrimHeap []PrimItem
func (h PrimHeap) Len() int           { return len(h) }
func (h PrimHeap) Less(i, j int) bool { return h[i].key < h[j].key }
func (h PrimHeap) Swap(i, j int)      { h[i], h[j] = h[j], h[i] }
func (h *PrimHeap) Push(x interface{}) { *h = append(*h, x.(PrimItem)) }
func (h *PrimHeap) Pop() interface{} {
	old := *h; n := len(old); x := old[n-1]; *h = old[0:n-1]; return x
}

// Prim 普里姆算法，返回MST的父节点映射和总权重
func (g *Graph) Prim(start int) (map[int]int, int) {
	key := make(map[int]int)
	parent := make(map[int]int)
	inMST := make(map[int]bool)
	for v := range g.vertices {
		key[v] = math.MaxInt32
		parent[v] = -1
	}
	key[start] = 0
	h := &PrimHeap{}
	heap.Init(h)
	heap.Push(h, PrimItem{key: 0, vertex: start})
	totalWeight := 0
	for h.Len() > 0 {
		item := heap.Pop(h).(PrimItem)
		u := item.vertex
		if inMST[u] { continue }
		inMST[u] = true
		totalWeight += key[u]
		for _, edge := range g.adjacency[u] {
			v := edge.to
			if !inMST[v] && edge.weight < key[v] {
				key[v] = edge.weight
				parent[v] = u
				heap.Push(h, PrimItem{key: edge.weight, vertex: v})
			}
		}
	}
	return parent, totalWeight
}

// ---------------------------- 4. 最小生成树：Kruskal算法 ----------------------------

type UnionFind struct {
	parent []int
	rank   []int
}
func NewUnionFind(size int) *UnionFind {
	p := make([]int, size)
	r := make([]int, size)
	for i := range p { p[i] = i; r[i] = 1 }
	return &UnionFind{parent: p, rank: r}
}
func (uf *UnionFind) Find(x int) int {
	if uf.parent[x] != x {
		uf.parent[x] = uf.Find(uf.parent[x])
	}
	return uf.parent[x]
}
func (uf *UnionFind) Union(x, y int) bool {
	rx, ry := uf.Find(x), uf.Find(y)
	if rx == ry { return false }
	if uf.rank[rx] < uf.rank[ry] {
		uf.parent[rx] = ry
	} else if uf.rank[rx] > uf.rank[ry] {
		uf.parent[ry] = rx
	} else {
		uf.parent[ry] = rx
		uf.rank[rx]++
	}
	return true
}

type KruskalEdge struct { from, to, weight int }
func (g *Graph) Kruskal() ([]KruskalEdge, int) {
	edges := []KruskalEdge{}
	for from, adj := range g.adjacency {
		for _, e := range adj {
			edges = append(edges, KruskalEdge{from: from, to: e.to, weight: e.weight})
		}
	}
	for i := 0; i < len(edges); i++ {
		for j := i+1; j < len(edges); j++ {
			if edges[i].weight > edges[j].weight {
				edges[i], edges[j] = edges[j], edges[i]
			}
		}
	}
	maxV := 0
	for v := range g.vertices { if v > maxV { maxV = v } }
	uf := NewUnionFind(maxV+1)
	mst := []KruskalEdge{}
	total := 0
	for _, e := range edges {
		if uf.Union(e.from, e.to) {
			mst = append(mst, e)
			total += e.weight
			if len(mst) == len(g.vertices)-1 { break }
		}
	}
	return mst, total
}

// ---------------------------- 5. 最短路径：Dijkstra算法 ----------------------------

type DijkstraItem struct { distance int; vertex int }
type DijkstraHeap []DijkstraItem
func (h DijkstraHeap) Len() int            { return len(h) }
func (h DijkstraHeap) Less(i, j int) bool  { return h[i].distance < h[j].distance }
func (h DijkstraHeap) Swap(i, j int)       { h[i], h[j] = h[j], h[i] }
func (h *DijkstraHeap) Push(x interface{}) { *h = append(*h, x.(DijkstraItem)) }
func (h *DijkstraHeap) Pop() interface{} {
	old := *h; n := len(old); x := old[n-1]; *h = old[0:n-1]; return x
}
func (g *Graph) Dijkstra(start int) (map[int]int, map[int]int) {
	dist := make(map[int]int)
	parent := make(map[int]int)
	visited := make(map[int]bool)
	for v := range g.vertices {
		dist[v] = math.MaxInt32
		parent[v] = -1
	}
	dist[start] = 0
	pq := &DijkstraHeap{}
	heap.Init(pq)
	heap.Push(pq, DijkstraItem{distance: 0, vertex: start})
	for pq.Len() > 0 {
		item := heap.Pop(pq).(DijkstraItem)
		u := item.vertex
		if visited[u] { continue }
		visited[u] = true
		for _, edge := range g.adjacency[u] {
			v := edge.to
			if dist[u]+edge.weight < dist[v] {
				dist[v] = dist[u] + edge.weight
				parent[v] = u
				heap.Push(pq, DijkstraItem{distance: dist[v], vertex: v})
			}
		}
	}
	return dist, parent
}

// ---------------------------- 6. 最短路径：Floyd算法 ----------------------------
func (g *Graph) Floyd() [][]int {
	maxV := 0
	for v := range g.vertices { if v > maxV { maxV = v } }
	n := maxV + 1
	dist := make([][]int, n)
	for i := range dist {
		dist[i] = make([]int, n)
		for j := range dist[i] {
			if i == j { dist[i][j] = 0 } else { dist[i][j] = math.MaxInt32 }
		}
	}
	for from, edges := range g.adjacency {
		for _, e := range edges {
			if e.weight < dist[from][e.to] { dist[from][e.to] = e.weight }
		}
	}
	for k := 0; k < n; k++ {
		for i := 0; i < n; i++ {
			for j := 0; j < n; j++ {
				if dist[i][k] != math.MaxInt32 && dist[k][j] != math.MaxInt32 && dist[i][k]+dist[k][j] < dist[i][j] {
					dist[i][j] = dist[i][k] + dist[k][j]
				}
			}
		}
	}
	return dist
}

// ---------------------------- 7. 拓扑排序 Kahn算法 ----------------------------
func (g *Graph) TopologicalSort() ([]int, bool) {
	inDeg := make(map[int]int)
	for v := range g.vertices { inDeg[v] = 0 }
	for _, adj := range g.adjacency {
		for _, e := range adj { inDeg[e.to]++ }
	}
	queue := []int{}
	for v, d := range inDeg { if d == 0 { queue = append(queue, v) } }
	res := []int{}
	for len(queue) > 0 {
		u := queue[0]; queue = queue[1:]
		res = append(res, u)
		for _, e := range g.adjacency[u] {
			inDeg[e.to]--
			if inDeg[e.to] == 0 { queue = append(queue, e.to) }
		}
	}
	if len(res) != len(g.vertices) { return res, false }
	return res, true
}

// ---------------------------- 8. AOE关键路径 ----------------------------
type AOEActivity struct { from, to, weight int }
func (g *Graph) CriticalPath() ([]AOEActivity, map[int]int, map[int]int) {
	topo, ok := g.TopologicalSort()
	if !ok {
		fmt.Println("AOE网有环，无法计算关键路径")
		return nil, nil, nil
	}
	ve := make(map[int]int)
	for _, u := range topo {
		ve[u] = 0
	}
	for _, u := range topo {
		for _, e := range g.adjacency[u] {
			if ve[u]+e.weight > ve[e.to] {
				ve[e.to] = ve[u] + e.weight
			}
		}
	}
	maxVe := 0
	for _, v := range topo { if ve[v] > maxVe { maxVe = ve[v] } }
	vl := make(map[int]int)
	for _, v := range topo { vl[v] = maxVe }
	for i := len(topo)-1; i >= 0; i-- {
		u := topo[i]
		for _, e := range g.adjacency[u] {
			if vl[e.to]-e.weight < vl[u] {
				vl[u] = vl[e.to] - e.weight
			}
		}
	}
	critical := []AOEActivity{}
	for u, edges := range g.adjacency {
		for _, e := range edges {
			if ve[u] == vl[e.to]-e.weight {
				critical = append(critical, AOEActivity{from: u, to: e.to, weight: e.weight})
			}
		}
	}
	return critical, ve, vl
}
