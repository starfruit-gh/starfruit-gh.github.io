顺序查找

// SequentialSearch 顺序查找，返回目标索引，未找到返回-1
func SequentialSearch(arr []int, target int) int {
    for i, v := range arr {
        if v == target {
            return i
        }
    }
    return -1
}

二分查找

// BinarySearchIterative 非递归二分查找，返回目标索引，未找到返回-1
func BinarySearchIterative(arr []int, target int) int {
    left, right := 0, len(arr)-1
    for left <= right {
        mid := left + (right-left)/2 // 避免溢出
        switch {
        case arr[mid] == target:
            return mid
        case arr[mid] < target:
            left = mid + 1
        default:
            right = mid - 1
        }
    }
    return -1
}

// BinarySearchRecursive 递归二分查找
func BinarySearchRecursive(arr []int, target, left, right int) int {
    if left > right {
        return -1
    }
    mid := left + (right-left)/2
    switch {
    case arr[mid] == target:
        return mid
    case arr[mid] < target:
        return BinarySearchRecursive(arr, target, mid+1, right)
    default:
        return BinarySearchRecursive(arr, target, left, mid-1)
    }
}

二叉搜索树

// TreeNode BST节点
type TreeNode struct {
    Val   int
    Left  *TreeNode
    Right *TreeNode
}

// BSTInsert 插入节点
func BSTInsert(root *TreeNode, val int) *TreeNode {
    if root == nil {
        return &TreeNode{Val: val}
    }
    if val < root.Val {
        root.Left = BSTInsert(root.Left, val)
    } else if val > root.Val {
        root.Right = BSTInsert(root.Right, val)
    }
    return root
}

// BSTSearch 查找节点
func BSTSearch(root *TreeNode, val int) *TreeNode {
    if root == nil || root.Val == val {
        return root
    }
    if val < root.Val {
        return BSTSearch(root.Left, val)
    }
    return BSTSearch(root.Right, val)
}

// BSTDelete 删除节点
func BSTDelete(root *TreeNode, val int) *TreeNode {
    if root == nil {
        return nil
    }
    // 1. 查找要删除的节点
    if val < root.Val {
        root.Left = BSTDelete(root.Left, val)
    } else if val > root.Val {
        root.Right = BSTDelete(root.Right, val)
    } else {
        // 2. 情况1：叶子节点或只有一个子节点
        if root.Left == nil {
            return root.Right
        } else if root.Right == nil {
            return root.Left
        }
        // 3. 情况2：有两个子节点，找右子树最小节点（后继）
        minNode := root.Right
        for minNode.Left != nil {
            minNode = minNode.Left
        }
        root.Val = minNode.Val
        root.Right = BSTDelete(root.Right, minNode.Val)
    }
    return root
}

平衡二叉树 (AVL)

// AVLTreeNode AVL节点
type AVLTreeNode struct {
    Val    int
    Left   *AVLTreeNode
    Right  *AVLTreeNode
    Height int // 节点高度
}

// getHeight 获取节点高度（空节点高度为-1）
func getHeight(node *AVLTreeNode) int {
    if node == nil {
        return -1
    }
    return node.Height
}

// max 辅助函数
func max(a, b int) int {
    if a > b {
        return a
    }
    return b
}

// rightRotate 右旋
func rightRotate(y *AVLTreeNode) *AVLTreeNode {
    x := y.Left
    T2 := x.Right

    // 旋转
    x.Right = y
    y.Left = T2

    // 更新高度
    y.Height = max(getHeight(y.Left), getHeight(y.Right)) + 1
    x.Height = max(getHeight(x.Left), getHeight(x.Right)) + 1

    return x
}

// leftRotate 左旋
func leftRotate(x *AVLTreeNode) *AVLTreeNode {
    y := x.Right
    T2 := y.Left

    // 旋转
    y.Left = x
    x.Right = T2

    // 更新高度
    x.Height = max(getHeight(x.Left), getHeight(x.Right)) + 1
    y.Height = max(getHeight(y.Left), getHeight(y.Right)) + 1

    return y
}

// getBalanceFactor 获取平衡因子
func getBalanceFactor(node *AVLTreeNode) int {
    if node == nil {
        return 0
    }
    return getHeight(node.Left) - getHeight(node.Right)
}

// AVLInsert 插入节点并平衡
func AVLInsert(node *AVLTreeNode, val int) *AVLTreeNode {
    // 1. 标准BST插入
    if node == nil {
        return &AVLTreeNode{Val: val, Height: 0}
    }
    if val < node.Val {
        node.Left = AVLInsert(node.Left, val)
    } else if val > node.Val {
        node.Right = AVLInsert(node.Right, val)
    } else {
        return node // 不允许重复值
    }

    // 2. 更新当前节点高度
    node.Height = 1 + max(getHeight(node.Left), getHeight(node.Right))

    // 3. 获取平衡因子，检查是否失衡
    balance := getBalanceFactor(node)

    // 4. 四种失衡情况处理
    // LL
    if balance > 1 && val < node.Left.Val {
        return rightRotate(node)
    }
    // RR
    if balance < -1 && val > node.Right.Val {
        return leftRotate(node)
    }
    // LR
    if balance > 1 && val > node.Left.Val {
        node.Left = leftRotate(node.Left)
        return rightRotate(node)
    }
    // RL
    if balance < -1 && val < node.Right.Val {
        node.Right = rightRotate(node.Right)
        return leftRotate(node)
    }

    return node
}

红黑树

const (
    RED   = true
    BLACK = false
)

// RBTreeNode 红黑树节点
type RBTreeNode struct {
    Val    int
    Left   *RBTreeNode
    Right  *RBTreeNode
    Parent *RBTreeNode
    Color  bool
}

// RBTree 红黑树
type RBTree struct {
    Root *RBTreeNode
    Nil  *RBTreeNode // 哨兵节点
}

// NewRBTree 初始化红黑树
func NewRBTree() *RBTree {
    nilNode := &RBTreeNode{Color: BLACK}
    return &RBTree{
        Root: nilNode,
        Nil:  nilNode,
    }
}

// leftRotate 左旋
func (t *RBTree) leftRotate(x *RBTreeNode) {
    y := x.Right
    x.Right = y.Left
    if y.Left != t.Nil {
        y.Left.Parent = x
    }
    y.Parent = x.Parent
    if x.Parent == t.Nil {
        t.Root = y
    } else if x == x.Parent.Left {
        x.Parent.Left = y
    } else {
        x.Parent.Right = y
    }
    y.Left = x
    x.Parent = y
}

// rightRotate 右旋
func (t *RBTree) rightRotate(y *RBTreeNode) {
    x := y.Left
    y.Left = x.Right
    if x.Right != t.Nil {
        x.Right.Parent = y
    }
    x.Parent = y.Parent
    if y.Parent == t.Nil {
        t.Root = x
    } else if y == y.Parent.Right {
        y.Parent.Right = x
    } else {
        y.Parent.Left = x
    }
    x.Right = y
    y.Parent = x
}

// insertFixup 插入后修复红黑树性质
func (t *RBTree) insertFixup(z *RBTreeNode) {
    for z.Parent.Color == RED {
        if z.Parent == z.Parent.Parent.Left {
            y := z.Parent.Parent.Right // 叔叔节点
            if y.Color == RED {
                // 情况1：叔叔红，变色
                z.Parent.Color = BLACK
                y.Color = BLACK
                z.Parent.Parent.Color = RED
                z = z.Parent.Parent
            } else {
                // 情况2：叔叔黑，z是右孩子，左旋
                if z == z.Parent.Right {
                    z = z.Parent
                    t.leftRotate(z)
                }
                // 情况3：叔叔黑，z是左孩子，右旋+变色
                z.Parent.Color = BLACK
                z.Parent.Parent.Color = RED
                t.rightRotate(z.Parent.Parent)
            }
        } else {
            // 对称情况
            y := z.Parent.Parent.Left
            if y.Color == RED {
                z.Parent.Color = BLACK
                y.Color = BLACK
                z.Parent.Parent.Color = RED
                z = z.Parent.Parent
            } else {
                if z == z.Parent.Left {
                    z = z.Parent
                    t.rightRotate(z)
                }
                z.Parent.Color = BLACK
                z.Parent.Parent.Color = RED
                t.leftRotate(z.Parent.Parent)
            }
        }
    }
    t.Root.Color = BLACK // 根节点始终黑
}

// Insert 插入节点
func (t *RBTree) Insert(val int) {
    // 1. 初始化新节点
    z := &RBTreeNode{
        Val:    val,
        Left:   t.Nil,
        Right:  t.Nil,
        Color:  RED,
        Parent: t.Nil,
    }

    // 2. 标准BST插入
    y := t.Nil
    x := t.Root
    for x != t.Nil {
        y = x
        if z.Val < x.Val {
            x = x.Left
        } else {
            x = x.Right
        }
    }
    z.Parent = y
    if y == t.Nil {
        t.Root = z
    } else if z.Val < y.Val {
        y.Left = z
    } else {
        y.Right = z
    }

    // 3. 修复红黑树性质
    if z.Parent == nil {
        z.Color = BLACK
        return
    }
    if z.Parent.Parent == nil {
        return
    }
    t.insertFixup(z)
}

B树

const M = 3 // B树阶数（最大子节点数）

// BTreeNode B树节点
type BTreeNode struct {
    Keys     []int
    Children []*BTreeNode
    IsLeaf   bool
}

// NewBTreeNode 创建新节点
func NewBTreeNode(isLeaf bool) *BTreeNode {
    return &BTreeNode{
        Keys:     make([]int, 0, M-1),
        Children: make([]*BTreeNode, 0, M),
        IsLeaf:   isLeaf,
    }
}

// BTree B树
type BTree struct {
    Root *BTreeNode
}

// searchHelper 辅助查找
func (n *BTreeNode) searchHelper(key int) *BTreeNode {
    i := 0
    for i < len(n.Keys) && key > n.Keys[i] {
        i++
    }
    if i < len(n.Keys) && key == n.Keys[i] {
        return n
    }
    if n.IsLeaf {
        return nil
    }
    return n.Children[i].searchHelper(key)
}

// Search 查找键
func (t *BTree) Search(key int) bool {
    if t.Root == nil {
        return false
    }
    return t.Root.searchHelper(key) != nil
}

// splitChild 分裂子节点
func (n *BTreeNode) splitChild(i int, y *BTreeNode) {
    z := NewBTreeNode(y.IsLeaf)
    mid := (M - 1) / 2 // 中间键索引

    // 复制y的后半部分键到z
    z.Keys = append(z.Keys, y.Keys[mid+1:]...)
    y.Keys = y.Keys[:mid]

    // 如果不是叶子，复制子节点
    if !y.IsLeaf {
        z.Children = append(z.Children, y.Children[mid+1:]...)
        y.Children = y.Children[:mid+1]
    }

    // 将z插入n的子节点
    n.Children = append(n.Children[:i+1], append([]*BTreeNode{z}, n.Children[i+1:]...)...)
    // 将中间键插入n
    n.Keys = append(n.Keys[:i], append([]int{y.Keys[mid]}, n.Keys[i:]...)...)
}

// insertNonFull 向非满节点插入键
func (n *BTreeNode) insertNonFull(key int) {
    i := len(n.Keys) - 1
    if n.IsLeaf {
        // 叶子节点，直接插入
        for i >= 0 && key < n.Keys[i] {
            i--
        }
        n.Keys = append(n.Keys[:i+1], append([]int{key}, n.Keys[i+1:]...)...)
    } else {
        // 非叶子节点，找到子节点
        for i >= 0 && key < n.Keys[i] {
            i--
        }
        i++
        // 如果子节点满了，先分裂
        if len(n.Children[i].Keys) == M-1 {
            n.splitChild(i, n.Children[i])
            if key > n.Keys[i] {
                i++
            }
        }
        n.Children[i].insertNonFull(key)
    }
}

// Insert 插入键
func (t *BTree) Insert(key int) {
    if t.Root == nil {
        t.Root = NewBTreeNode(true)
        t.Root.Keys = append(t.Root.Keys, key)
        return
    }
    // 如果根节点满了，先分裂
    if len(t.Root.Keys) == M-1 {
        newRoot := NewBTreeNode(false)
        newRoot.Children = append(newRoot.Children, t.Root)
        newRoot.splitChild(0, t.Root)
        t.Root = newRoot
    }
    t.Root.insertNonFull(key)
}

B+树

const BPlusM = 3 // B+树阶数

// BPlusTreeNode B+树节点
type BPlusTreeNode struct {
    Keys     []int
    Children []*BPlusTreeNode
    Next     *BPlusTreeNode // 叶子节点链表指针
    IsLeaf   bool
}

// NewBPlusTreeNode 创建新节点
func NewBPlusTreeNode(isLeaf bool) *BPlusTreeNode {
    return &BPlusTreeNode{
        Keys:     make([]int, 0, BPlusM-1),
        Children: make([]*BPlusTreeNode, 0, BPlusM),
        IsLeaf:   isLeaf,
    }
}

// BPlusTree B+树
type BPlusTree struct {
    Root *BPlusTreeNode
    Head *BPlusTreeNode // 叶子节点链表头
}

// Search 查找键（仅在叶子节点）
func (t *BPlusTree) Search(key int) bool {
    if t.Root == nil {
        return false
    }
    node := t.Root
    // 找到叶子节点
    for !node.IsLeaf {
        i := 0
        for i < len(node.Keys) && key >= node.Keys[i] {
            i++
        }
        node = node.Children[i]
    }
    // 在叶子节点链表中查找
    for _, k := range node.Keys {
        if k == key {
            return true
        }
    }
    return false
}

// splitChild 分裂子节点
func (n *BPlusTreeNode) splitChild(i int, y *BPlusTreeNode) {
    z := NewBPlusTreeNode(y.IsLeaf)
    mid := (BPlusM - 1) / 2

    // 复制键和子节点
    if y.IsLeaf {
        // 叶子节点：复制后半部分键，维护链表
        z.Keys = append(z.Keys, y.Keys[mid:]...)
        y.Keys = y.Keys[:mid]
        z.Next = y.Next
        y.Next = z
    } else {
        // 非叶子节点：复制后半部分键和子节点
        z.Keys = append(z.Keys, y.Keys[mid+1:]...)
        y.Keys = y.Keys[:mid]
        z.Children = append(z.Children, y.Children[mid+1:]...)
        y.Children = y.Children[:mid+1]
    }

    // 插入z到n的子节点
    n.Children = append(n.Children[:i+1], append([]*BPlusTreeNode{z}, n.Children[i+1:]...)...)
    // 插入中间键到n（非叶子节点取y.Keys[mid]，叶子节点取z.Keys[0]）
    var insertKey int
    if y.IsLeaf {
        insertKey = z.Keys[0]
    } else {
        insertKey = y.Keys[mid]
    }
    n.Keys = append(n.Keys[:i], append([]int{insertKey}, n.Keys[i:]...)...)
}

// insertNonFull 向非满节点插入
func (n *BPlusTreeNode) insertNonFull(key int, isLeaf bool) {
    i := len(n.Keys) - 1
    if n.IsLeaf {
        // 叶子节点插入
        for i >= 0 && key < n.Keys[i] {
            i--
        }
        n.Keys = append(n.Keys[:i+1], append([]int{key}, n.Keys[i+1:]...)...)
    } else {
        // 非叶子节点找到子节点
        for i >= 0 && key < n.Keys[i] {
            i--
        }
        i++
        if len(n.Children[i].Keys) == BPlusM-1 {
            n.splitChild(i, n.Children[i])
            if key >= n.Keys[i] {
                i++
            }
        }
        n.Children[i].insertNonFull(key, isLeaf)
    }
}

// Insert 插入键
func (t *BPlusTree) Insert(key int) {
    if t.Root == nil {
        t.Root = NewBPlusTreeNode(true)
        t.Root.Keys = append(t.Root.Keys, key)
        t.Head = t.Root
        return
    }
    // 根节点满了分裂
    if len(t.Root.Keys) == BPlusM-1 {
        newRoot := NewBPlusTreeNode(false)
        newRoot.Children = append(newRoot.Children, t.Root)
        newRoot.splitChild(0, t.Root)
        t.Root = newRoot
    }
    t.Root.insertNonFull(key, true)
}

哈希表

// ListNode 链表节点
type ListNode struct {
    Key  int
    Val  int
    Next *ListNode
}

// HashTable 哈希表
type HashTable struct {
    Buckets  []*ListNode
    Size     int
    Capacity int
}

// NewHashTable 初始化哈希表
func NewHashTable(capacity int) *HashTable {
    return &HashTable{
        Buckets:  make([]*ListNode, capacity),
        Capacity: capacity,
    }
}

// hashFunc 哈希函数（取模）
func (ht *HashTable) hashFunc(key int) int {
    return key % ht.Capacity
}

// Put 插入键值对
func (ht *HashTable) Put(key, val int) {
    index := ht.hashFunc(key)
    // 检查是否已存在，存在则更新
    current := ht.Buckets[index]
    for current != nil {
        if current.Key == key {
            current.Val = val
            return
        }
        current = current.Next
    }
    // 不存在则头插
    newNode := &ListNode{Key: key, Val: val, Next: ht.Buckets[index]}
    ht.Buckets[index] = newNode
    ht.Size++
}

// Get 查找键对应的值
func (ht *HashTable) Get(key int) (int, bool) {
    index := ht.hashFunc(key)
    current := ht.Buckets[index]
    for current != nil {
        if current.Key == key {
            return current.Val, true
        }
        current = current.Next
    }
    return 0, false
}

// Remove 删除键
func (ht *HashTable) Remove(key int) {
    index := ht.hashFunc(key)
    current := ht.Buckets[index]
    var prev *ListNode
    for current != nil {
        if current.Key == key {
            if prev == nil {
                ht.Buckets[index] = current.Next
            } else {
                prev.Next = current.Next
            }
            ht.Size--
            return
        }
        prev = current
        current = current.Next
    }
}
