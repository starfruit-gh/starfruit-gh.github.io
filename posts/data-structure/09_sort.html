直接插入排序

func StraightInsertionSort(arr []int) {
    n := len(arr)
    for i := 1; i < n; i++ {
        key := arr[i] // 当前待插入元素
        j := i - 1    // 已排序部分的最后一个元素索引

        // 向前遍历，移动比key大的元素
        for j >= 0 && arr[j] > key {
            arr[j+1] = arr[j]
            j--
        }
        arr[j+1] = key // 插入到正确位置
    }
}

折半插入排序

func BinaryInsertionSort(arr []int) {
	n := len(arr)
	for i := 1; i < n; i++ {
		key := arr[i]
		low, high := 0, i-1

		// 二分查找插入位置（low为最终插入点）
		for low <= high {
			mid := (low + high) / 2
			if arr[mid] > key {
				high = mid - 1
			} else {
				low = mid + 1
			}
		}

		// 移动元素并插入
		for j := i - 1; j >= low; j-- {
			arr[j+1] = arr[j]
		}
		arr[low] = key
	}
}

希尔排序

func ShellSort(arr []int) {
	n := len(arr)
	// 希尔增量：初始为n/2，每次减半
	for gap := n / 2; gap > 0; gap /= 2 {
		// 对每个gap进行插入排序
		for i := gap; i < n; i++ {
			key := arr[i]
			j := i - gap
			for j >= 0 && arr[j] > key {
				arr[j+gap] = arr[j]
				j -= gap
			}
			arr[j+gap] = key
		}
	}
}

冒泡排序

func BubbleSort(arr []int) {
	n := len(arr)
	for i := 0; i < n-1; i++ {
		swapped := false // 优化标志：本轮是否发生交换
		// 每轮将最大元素移到末尾
		for j := 0; j < n-1-i; j++ {
			if arr[j] > arr[j+1] {
				arr[j], arr[j+1] = arr[j+1], arr[j]
				swapped = true
			}
		}
		if !swapped { // 无交换则已排序，提前退出
			break
		}
	}
}

快速排序

func QuickSort(arr []int) {
	if len(arr) <= 1 {
		return
	}
	pivotIndex := partition(arr)
	QuickSort(arr[:pivotIndex])
	QuickSort(arr[pivotIndex+1:])
}

// 分区函数：返回基准元素的最终位置
func partition(arr []int) int {
	n := len(arr)
	pivot := arr[n-1] // 选最后一个元素为基准
	i := 0             // 小于pivot的元素的最后一个索引

	for j := 0; j < n-1; j++ {
		if arr[j] <= pivot {
			arr[i], arr[j] = arr[j], arr[i]
			i++
		}
	}
	arr[i], arr[n-1] = arr[n-1], arr[i] // 基准归位
	return i
}

// 优化版快速排序：随机选择基准，避免最坏情况
func QuickSortOptimized(arr []int) {
	if len(arr) <= 1 {
		return
	}
	pivotIndex := partitionOptimized(arr)
	QuickSortOptimized(arr[:pivotIndex])
	QuickSortOptimized(arr[pivotIndex+1:])
}

func partitionOptimized(arr []int) int {
	n := len(arr)
	// 随机选择基准，与最后一个元素交换
	randIndex := rand.Intn(n)
	arr[randIndex], arr[n-1] = arr[n-1], arr[randIndex]
	return partition(arr) // 复用基础分区逻辑
}

简单选择排序

func SimpleSelectionSort(arr []int) {
	n := len(arr)
	for i := 0; i < n-1; i++ {
		minIndex := i // 假设当前i为最小元素索引
		// 遍历未排序部分，找最小元素
		for j := i + 1; j < n; j++ {
			if arr[j] < arr[minIndex] {
				minIndex = j
			}
		}
		// 交换到已排序部分末尾
		if minIndex != i {
			arr[i], arr[minIndex] = arr[minIndex], arr[i]
		}
	}
}

堆排序

func HeapSort(arr []int) {
	n := len(arr)
	// 1. 建大顶堆：从最后一个非叶子节点开始调整
	for i := n/2 - 1; i >= 0; i-- {
		heapify(arr, n, i)
	}
	// 2. 逐个交换堆顶（最大元素）与末尾元素，调整堆
	for i := n - 1; i > 0; i-- {
		arr[0], arr[i] = arr[i], arr[0]
		heapify(arr, i, 0) // 调整前i个元素的堆
	}
}

// 调整堆：n为堆大小，i为待调整节点索引
func heapify(arr []int, n, i int) {
	largest := i       // 假设i为最大节点
	left := 2*i + 1    // 左子节点
	right := 2*i + 2   // 右子节点

	// 找最大的节点
	if left < n && arr[left] > arr[largest] {
		largest = left
	}
	if right < n && arr[right] > arr[largest] {
		largest = right
	}

	// 若最大节点不是i，交换并继续调整
	if largest != i {
		arr[i], arr[largest] = arr[largest], arr[i]
		heapify(arr, n, largest)
	}
}

归并排序

func MergeSort(arr []int) []int {
	if len(arr) <= 1 {
		return arr
	}
	// 拆分
	mid := len(arr) / 2
	left := MergeSort(arr[:mid])
	right := MergeSort(arr[mid:])
	// 合并
	return merge(left, right)
}

// 合并两个有序数组
func merge(left, right []int) []int {
	result := make([]int, 0, len(left)+len(right))
	i, j := 0, 0

	// 按顺序合并
	for i < len(left) && j < len(right) {
		if left[i] <= right[j] {
			result = append(result, left[i])
			i++
		} else {
			result = append(result, right[j])
			j++
		}
	}
	// 加入剩余元素
	result = append(result, left[i:]...)
	result = append(result, right[j:]...)
	return result
}

基数排序

func RadixSort(arr []int) {
	if len(arr) <= 1 {
		return
	}
	// 1. 找最大值，确定最大位数
	maxVal := arr[0]
	for _, num := range arr {
		if num > maxVal {
			maxVal = num
		}
	}
	// 2. 按位处理（从低位到高位）
	for exp := 1; maxVal/exp > 0; exp *= 10 {
		countSort(arr, exp)
	}
}

// 计数排序变种：按当前位（exp）排序
func countSort(arr []int, exp int) {
	n := len(arr)
	output := make([]int, n) // 输出数组
	count := make([]int, 10) // 0-9的桶

	// 1. 统计每个桶的元素个数
	for i := 0; i < n; i++ {
		digit := (arr[i] / exp) % 10
		count[digit]++
	}
	// 2. 计算前缀和，确定元素在output中的位置
	for i := 1; i < 10; i++ {
		count[i] += count[i-1]
	}
	// 3. 从后往前遍历（保证稳定性）
	for i := n - 1; i >= 0; i-- {
		digit := (arr[i] / exp) % 10
		output[count[digit]-1] = arr[i]
		count[digit]--
	}
	// 4. 复制回原数组
	copy(arr, output)
}
