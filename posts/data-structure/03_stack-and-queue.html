（三）栈和队列

栈的定义

栈（Stack）：限定仅在表尾进行插入或删除操作的线性表。

  • 不含任何元素的空表为空栈。
  • 允许插入和删除的一端为栈顶（Top），另一端为栈底（Bottom）。
  • 插入元素到栈顶的操作，称为入栈（Push）。
  • 从栈顶删除元素的操作，称为出栈（Pop）。

栈的特点

后进先出（LIFO）

举一个栈的例子：

假设有 3 个整型数字元素 1、2、3 依次进栈，会有哪些出栈次序呢?

  • 第一种：1、2、3 进，再 3、2、1 出。这是最简单的最好理解的一种，出栈次序为 321。
  • 第二种：1 进，1 出，2 进，2 出，3 进，3 出。也就是进一个就出一个，出栈次序为 123。
  • 第三种：1 进，2 进，2 出，1 出，3 进，3 出。出栈次序为 213。
  • 第四种：1 进，1 出，2 进，3 进，3 出，2 出。出栈次序为 132。
  • 第五种：1 进，2 进，2 出，3 进，3 出，1 出。出栈次序为 231。

有没有可能是 312 这样的次序出栈呢？

答案是肯定不会。因为 3 先出栈，就意味着 3 曾经进栈，既然 3 都进栈了，那也就意味着 1 和 2 已经进栈了，此时，2 一定是在 1 的上面，就是更接近栈顶，那么出栈只可能是 321，不然不满足 1、2、3 依次进栈的要求，所以此时不会发生 1 比 2 先出栈的情况。

    数据关系：R = {<aᵢ₋₁, aᵢ> | aᵢ₋₁, aᵢ ∈ D, i = 2,...,n}

栈的顺序存储结构

顺序栈的定义

采用顺序存储结构的栈称为顺序栈。它用一组地址连续的存储单元依次存放自栈底到栈顶的数据元素。

  • 附设 top 指针，指示栈顶元素在顺序栈中的位置。（但是，为了方便操作，top 通常指向栈顶元素的下一个位置，而非栈顶元素本身。）
  • 另设 base 指针，指示栈底元素在顺序栈中的位置。
  • 并用一个整型变量 stacksize 表示顺序栈的最大容量。

顺序栈的代码实现

定义

typedef struct {
    // 1. 栈底指针，指向动态分配的数组首地址
    ElemType *base;
    // 2. 栈顶指针，指向栈顶元素的下一个位置
    ElemType *top;
    // 3. 栈的最大容量
    int stacksize;
} SeqStack;

初始化

// InitStack 初始化顺序栈
bool InitStack(SeqStack &S) {
    // 1. 新建一个 ElemType 类型的一维数组，并为该数组申请一块指定大小的内存空间，让栈底指针 base 指向该数组
    S.base = new ElemType[MAXSIZE];
    // 2. 将栈顶指针 top 初始化为栈底指针，表示空栈
    S.top = S.base;
    // 3. 将栈的最大容量设置为 MAXSIZE
    S.stacksize = MAXSIZE;
    return true;
}

销毁

// DestroyStack 销毁顺序栈
void DestroyStack(SeqStack &S) {
    // 1. 判断栈底指针是否为空，为空则没有需要销毁的空间，直接返回
    if (S.base) {
        // 2. 释放栈底指针指向的数组占用的内存空间
        delete[] S.base;
        // 3. 将栈的最大容量置为 0
        S.stacksize = 0;
        // 4. 将栈底指针和栈顶指针都置为 NULL
        S.base = S.top = NULL;
    }
}

清空

// ClearStack 清空顺序栈
void ClearStack(SeqStack &S) {
    // 1. 判断栈底指针是否为空，为空则无需清空
    if (S.base) 
        // 2. 将栈顶指针重置为栈底指针，表示空栈
        S.top = S.base;
}

判空

// StackEmpty 判断顺序栈是否为空
bool StackEmpty(const SeqStack &S) {
    // 直接判断栈顶指针是否等于栈底指针返回结果即可
    if (S.top == S.base) return true;
    else return false;
}

求表长

// StackLength 返回顺序栈的长度
int StackLength(const SeqStack &S) {
    // 直接返回栈顶指针与栈底指针的差值，即为栈中元素个数
    return S.top - S.base;
}

取栈顶元素

// GetTop 获取顺序栈的栈顶元素，并用 e 返回其值
bool GetTop(const SeqStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 false
    if (S.top == S.base) return false;
    // 2. 将栈顶元素（栈顶指针前一个位置的元素）的值赋给 e
    e = *(S.top - 1);
    return true;
}

入栈

// Push 向顺序栈中插入元素 e 作为新的栈顶元素
bool Push(SeqStack &S, ElemType e) {
    // 1. 判断栈是否已满，若满则返回 false
    if (S.top - S.base == S.stacksize) return false;
    // 2. 将元素 e 放入栈顶指针指向的位置
    *S.top = e;
    // 3. 栈顶指针上移一位
    S.top++;
    return true;
}

出栈

// Pop 删除顺序栈的栈顶元素，并用 e 返回其值
bool Pop(SeqStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 false
    if (S.top == S.base) return false;
    // 2. 栈顶指针下移一位，指向栈顶元素
    S.top--;
    // 3. 将栈顶元素的值赋给 e
    e = *S.top;
    return true;
}

遍历

// StackTraverse 遍历顺序栈（从栈底到栈顶依次访问每个元素）
void StackTraverse(const SeqStack &S) {
    // 1. 定义临时指针，从栈底开始遍历
    ElemType *p = S.base;
    // 2. 循环遍历，直到临时指针到达栈顶位置
    while (p != S.top) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", *p);
        // 4. 指针后移，访问下一个元素
        p++;
    }
    // 换行，格式化输出
    printf("\n");
}

// StackTraverse 遍历顺序栈（自定义访问函数，更通用）
void StackTraverse(const SeqStack &S, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从栈底开始遍历
    ElemType *p = S.base;
    // 2. 循环遍历所有栈元素
    while (p != S.top) {
        // 3. 调用自定义函数访问元素
        visit(*p);
        // 4. 指针后移
        p++;
    }
}

栈的链式存储结构

链栈的定义

采用链式存储结构的栈称为链栈。一般采用单链表且不需要头结点，top 指针指向首元节点。

链栈的代码实现

定义

typedef struct StackNode{
    // 1. 结点数据域，存储栈元素
    ElemType data;
    // 2. 结点指针域，指向后继结点
    struct StackNode *next;
} StackNode, *LinkStack;

初始化

// InitStack 初始化链栈
void InitStack(LinkStack &S) {
    // 将栈顶指针置为 NULL，表示空栈
    S.top = NULL;
}

销毁

// DestroyStack 销毁链栈
void DestroyStack(LinkStack &S) {
    // 1. 定义临时指针，用于遍历释放结点
    StackNode *p;
    // 2. 循环释放所有结点，直到栈顶指针为 NULL
    while (S.top) {
        // 3. 临时指针指向当前栈顶结点
        p = S.top;
        // 4. 栈顶指针下移，指向下一个结点
        S.top = S.top->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
}

清空

// ClearStack 清空链栈
void ClearStack(LinkStack &S) {
    // 1. 链栈的清空操作与销毁操作逻辑一致，释放所有结点并将栈顶指针置为 NULL
    // 定义临时指针，用于遍历释放结点
    StackNode *p;
    // 2. 循环释放所有结点，直到栈顶指针为 NULL
    while (S.top) {
        // 3. 临时指针指向当前栈顶结点
        p = S.top;
        // 4. 栈顶指针下移，指向下一个结点
        S.top = S.top->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
}

判空

// StackEmpty 判断链栈是否为空
bool StackEmpty(const LinkStack &S) {
    // 直接判断栈顶指针是否为 NULL 返回结果即可
    if (S.top == NULL) return true;
    else return false;
}

求表长

// StackLength 返回链栈的长度
int StackLength(const LinkStack &S) {
    // 1. 定义计数器，初始化为 0
    int len = 0;
    // 2. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 3. 循环遍历所有结点
    while (p) {
        // 4. 计数器加 1
        len++;
        // 5. 指针后移，访问下一个结点
        p = p->next;
    }
    // 6. 返回栈的长度
    return len;
}

取栈顶元素

// GetTop 获取链栈的栈顶元素，并用 e 返回其值
bool GetTop(const LinkStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 false
    if (S.top == NULL) return false;
    // 2. 将栈顶结点的数据域的值赋给 e
    e = S.top->data;
    return true;
}

入栈

// Push 向链栈中插入元素 e 作为新的栈顶元素
bool Push(LinkStack &S, ElemType e) {
    // 1. 新建一个结点，为其分配内存空间
    StackNode *p = new StackNode;
    // 2. 判断新结点是否分配成功，失败则退出
    if (!p) exit(OVERFLOW);
    // 3. 将元素 e 赋给新结点的数据域
    p->data = e;
    // 4. 将新结点的指针域指向当前的栈顶结点
    p->next = S.top;
    // 5. 将栈顶指针更新为新结点
    S.top = p;
    return true;
}

出栈

// Pop 删除链栈的栈顶元素，并用 e 返回其值
bool Pop(LinkStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 false
    if (S.top == NULL) return false;
    // 2. 定义临时指针，指向当前栈顶结点
    StackNode *p = S.top;
    // 3. 将栈顶结点的数据域的值赋给 e
    e = p->data;
    // 4. 将栈顶指针下移，指向下一个结点
    S.top = S.top->next;
    // 5. 释放临时指针指向的结点
    delete p;
    return true;
}

遍历

// StackTraverse 遍历链栈（从栈顶到栈底依次访问每个元素）
void StackTraverse(const LinkStack &S) {
    // 1. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 2. 循环遍历所有结点
    while (p) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", p->data);
        // 4. 指针后移，访问下一个结点
        p = p->next;
    }
    // 换行，格式化输出
    printf("\n");
}

// StackTraverse 遍历链栈（自定义访问函数，更通用）
void StackTraverse(const LinkStack &S, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 2. 循环遍历所有结点
    while (p) {
        // 3. 调用自定义函数访问元素
        visit(p->data);
        // 4. 指针后移
        p = p->next;
    }
}

队列的定义

队列（Queue）：只允许在表的一端进行插入操作，而在另一端进行删除操作的线性表。

  • 不含任何元素的空表为空队列。
  • 允许插入的一端为队尾（Rear），允许删除的一端为队头（Front）。
  • 插入元素到队尾的操作，称为入队（EnQueue）。
  • 从队头删除元素的操作，称为出队（DeQueue）。

队列的特点

先进先出（FIFO）

队列的抽象数据类型

ADT Queue {
    数据对象：D = {aᵢ | aᵢ ∈ ElemSet, i = 1,2,...,n, n ≥ 0}
    数据关系：R = {<aᵢ₋₁, aᵢ> | aᵢ₋₁, aᵢ ∈ D, i = 2,...,n}
    基本操作：
        // 1. 初始化
        InitQueue(&Q)
        // 2. 销毁
        DestroyQueue(&Q)
        // 3. 清空
        ClearQueue(&Q)
        // 4. 判空
        QueueEmpty(Q)
        // 5. 求表长
        QueueLength(Q)
        // 6. 取队头元素
        GetHead(Q)
        // 7. 入队
        EnQueue(&Q, e)
        // 8. 出队
        DeQueue(&Q, &e)
        // 9. 遍历
        QueueTraverse(Q)
} ADT Queue

队列的顺序存储结构

顺序队列的定义

采用顺序存储结构的队列称为顺序队列。它用一组地址连续的存储单元依次存放自队头到队尾的数据元素。

  • 附设 front 指针，指示队头元素在顺序队列中的位置。
  • 另设 rear 指针，指示队尾元素在顺序队列中的位置。（rear 通常指向队尾元素的下一个位置，而非队尾元素本身。）

顺序队列的假溢出问题

当顺序队列的 rear 指针到达队列的末尾时，则无法再插入新的元素，否则会出现溢出现象。但是此时队列中可能仍然还有空闲空间，这就是顺序队列的假溢出问题。

循环队列的定义

循环队列（Circular Queue）：通过取模运算将普通顺序队列的首尾逻辑上连接起来，形成一个环状结构。从而避免了普通顺序队列的假溢出问题。

在循环队列中，当 rear 到达数组末尾时，下一步会绕回到数组开头（通过取模运算实现），实现空间的循环利用。

循环队列存在的一个问题是如何区分队列为空和队列已满的情况。因为当队头指针和队尾指针相等时，可能是队列为空，也可能是队列已满。

解决循环队列判空和判满问题的一个比较好的方法就是：在队列中预留一个位置，front = rear 时，队列为空；front = (rear + 1) % MAXQSIZE 时，队列已满。

循环队列的代码实现

定义

typedef struct {
    // 1. 动态分配的数组首地址，存储队列所有元素
    ElemType *base;
    // 2. 队头指针，指向队列的头元素
    int front;
    // 3. 队尾指针，指向队列尾元素的下一个位置
    int rear;
} SeqQueue;

初始化

// InitQueue 初始化顺序循环队列
bool InitQueue(SeqQueue &Q) {
    // 1. 新建一个 ElemType 类型的一维数组，申请指定大小的内存空间
    Q.base = new ElemType[MAXQSIZE];
    // 2. 队头和队尾指针都置为 0，表示空队列
    Q.front = Q.rear = 0;
    return true;
}

销毁

// DestroyQueue 销毁顺序循环队列
void DestroyQueue(SeqQueue &Q) {
    // 1. 判断数组指针是否为空，不为空则释放其占用的内存
    if (Q.base) {
        delete[] Q.base;
        // 2. 将数组指针置为 NULL
        Q.base = NULL;
    }
    // 3. 队头和队尾指针都置为 0
    Q.front = Q.rear = 0;
}

清空

// ClearQueue 清空顺序循环队列
void ClearQueue(SeqQueue &Q) {
    // 直接将队头和队尾指针都置为 0 即可
    Q.front = Q.rear = 0;
}

判空

// QueueEmpty 判断顺序循环队列是否为空
bool QueueEmpty(SeqQueue &Q) {
    // 直接判断队头指针是否等于队尾指针返回结果即可
    if (Q.front == Q.rear) return true;
    else return false;
}

求表长

// QueueLength 返回顺序循环队列的长度
int QueueLength(const SeqQueue &Q) {
    // 1. 循环队列长度计算公式：(队尾指针 - 队头指针 + 最大容量) % 最大容量
    return (Q.rear - Q.front + MAXQSIZE) % MAXQSIZE;
}

取队头元素

// GetHead 获取顺序循环队列的队头元素，并用 e 返回其值
bool GetHead(const SeqQueue &Q, ElemType &e) {
    // 1. 判断队列是否为空，为空则返回 false
    if (Q.front == Q.rear) return false;
    // 2. 将队头元素的值赋给 e
    e = Q.base[Q.front];
    return true;
}

入队

// EnQueue 向顺序循环队列中插入元素 e 作为新的队尾元素
bool EnQueue(SeqQueue &Q, ElemType e) {
    // 1. 判断队列是否已满，满则返回 false
    if ((Q.rear + 1) % MAXQSIZE == Q.front) return false;
    // 2. 将元素 e 放入队尾指针指向的位置
    Q.base[Q.rear] = e;
    // 3. 队尾指针循环后移一位
    Q.rear = (Q.rear + 1) % MAXQSIZE;
    return true;
}

出队

// DeQueue 删除顺序循环队列的队头元素，并用 e 返回其值
bool DeQueue(SeqQueue &Q, ElemType &e) {
    // 1. 判断队列是否为空，为空则返回 false
    if (Q.front == Q.rear) return false;
    // 2. 将队头元素的值赋给 e
    e = Q.base[Q.front];
    // 3. 队头指针循环后移一位
    Q.front = (Q.front + 1) % MAXQSIZE;
    return true;
}

遍历

// QueueTraverse 遍历顺序循环队列（从队头到队尾依次访问每个元素）
void QueueTraverse(SeqQueue &Q) {
    // 1. 定义临时指针，从队头开始遍历
    int p = Q.front;
    // 2. 循环遍历，直到临时指针到达队尾位置
    while (p != Q.rear) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", Q.base[p]);
        // 4. 临时指针循环后移一位
        p = (p + 1) % MAXQSIZE;
    }
    // 换行，格式化输出
    printf("\n");
}

// QueueTraverse 遍历顺序循环队列（自定义访问函数，更通用）
void QueueTraverse(SeqQueue &Q, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从队头开始遍历
    int p = Q.front;
    // 2. 循环遍历所有队列元素
    while (p != Q.rear) {
        // 3. 调用自定义函数访问元素
        visit(Q.base[p]);
        // 4. 临时指针循环后移一位
        p = (p + 1) % MAXQSIZE;
    }
}

队列的链式存储结构

链式队列的定义

采用链式存储结构的队列称为链式队列。一般采用单链表且带有头结点，front 指针指向头结点，rear 指针指向尾结点。

链式队列的代码实现

定义

// 链队结点结构
typedef struct QNode {
    // 1. 结点数据域，存储队列元素
    ElemType data;
    // 2. 结点指针域，指向后继结点
    struct QNode *next;
} QNode;

// 链队结构
typedef struct {
    // 1. 队头指针，指向头结点
    QNode *front;
    // 2. 队尾指针，指向队尾结点
    QNode *rear;
} LinkQueue;

初始化

// InitQueue 初始化链队（带头结点）
bool InitQueue(LinkQueue &Q) {
    // 1. 新建一个头结点，为其分配内存空间
    Q.front = Q.rear = new QNode;
    // 2. 判断头结点是否分配成功，失败则退出
    if (!Q.front) exit(OVERFLOW);
    // 3. 将头结点的指针域置为 NULL，表示空队列
    Q.front->next = NULL;
    return true;
}

销毁

// DestroyQueue 销毁链队
bool DestroyQueue(LinkQueue &Q) {
    // 1. 定义临时指针，用于遍历释放结点
    QNode *p;
    // 2. 循环释放所有结点，包括头结点
    while (Q.front) {
        // 3. 临时指针指向当前头结点
        p = Q.front;
        // 4. 队头指针后移，指向下一个结点
        Q.front = Q.front->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
    // 6. 释放后头结点指针已为 NULL，队尾指针也置为 NULL
    Q.rear = NULL;
    return true;
}

清空

// ClearQueue 清空链队（保留头结点）
bool ClearQueue(LinkQueue &Q) {
    // 1. 定义临时指针，指向队头结点的下一个结点（第一个数据结点）
    QNode *p = Q.front->next;
    // 2. 定义临时指针，用于释放结点
    QNode *q;
    // 3. 循环释放所有数据结点
    while (p) {
        // 4. 临时指针 q 指向当前要释放的结点
        q = p;
        // 5. 指针 p 后移，指向下一个结点
        p = p->next;
        // 6. 释放临时指针 q 指向的结点
        delete q;
    }
    // 7. 队头结点的指针域置为 NULL
    Q.front->next = NULL;
    // 8. 队尾指针重新指向头结点
    Q.rear = Q.front;
    return true;
}

判空

// QueueEmpty 判断链队是否为空
bool QueueEmpty(const LinkQueue &Q) {
    // 1. 判断队头指针的下一个结点是否为 NULL，是则返回 true，否则返回 false
    if (Q.front->next == NULL)
        return true;
    else
        return false;
}

求表长

// QueueLength 返回链队的长度
int QueueLength(const LinkQueue &Q) {
    // 1. 定义计数器，初始化为 0
    int len = 0;
    // 2. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 3. 循环遍历所有数据结点
    while (p) {
        // 4. 计数器加 1
        len++;
        // 5. 指针后移，访问下一个结点
        p = p->next;
    }
    // 6. 返回队列的长度
    return len;
}

取队头元素

// GetHead 获取链队的队头元素，并用 e 返回其值
bool GetHead(const LinkQueue &Q, ElemType &e) {
    // 1. 判断队列是否为空，为空则返回 false
    if (Q.front->next == NULL) return false;
    // 2. 将第一个数据结点的数据域的值赋给 e
    e = Q.front->next->data;
    return true;
}

入队

// EnQueue 向链队中插入元素 e 作为新的队尾元素
bool EnQueue(LinkQueue &Q, ElemType e) {
    // 1. 新建一个结点，为其分配内存空间
    QNode *p = new QNode;
    // 2. 判断新结点是否分配成功，失败则退出
    if (!p) exit(OVERFLOW);
    // 3. 将元素 e 赋给新结点的数据域
    p->data = e;
    // 4. 将新结点的指针域置为 NULL
    p->next = NULL;
    // 5. 将新结点插入到队尾结点的后面
    Q.rear->next = p;
    // 6. 将队尾指针更新为新结点
    Q.rear = p;
    return true;
}

出队

// DeQueue 删除链队的队头元素，并用 e 返回其值
bool DeQueue(LinkQueue &Q, ElemType &e) {
    // 1. 判断队列是否为空，为空则返回 false
    if (Q.front->next == NULL) return false;
    // 2. 定义临时指针，指向第一个数据结点（队头元素所在结点）
    QNode *p = Q.front->next;
    // 3. 将队头元素的数据域的值赋给 e
    e = p->data;
    // 4. 将头结点的指针域指向第二个数据结点
    Q.front->next = p->next;
    // 5. 判断删除的是否为队尾结点，若是则将队尾指针重新指向头结点
    if (Q.rear == p)
        Q.rear = Q.front;
    // 6. 释放临时指针指向的结点
    delete p;
    return true;
}

遍历

// QueueTraverse 遍历链队（从队头到队尾依次访问每个元素）
void QueueTraverse(const LinkQueue &Q) {
    // 1. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 2. 循环遍历所有数据结点
    while (p) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", p->data);
        // 4. 指针后移，访问下一个结点
        p = p->next;
    }
    // 换行，格式化输出
    printf("\n");
}

// QueueTraverse 遍历链队（自定义访问函数，更通用）
void QueueTraverse(const LinkQueue &Q, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 2. 循环遍历所有数据结点
    while (p) {
        // 3. 调用自定义函数访问元素
        visit(p->data);
        // 4. 指针后移
        p = p->next;
    }
}
