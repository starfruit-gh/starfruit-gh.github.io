（三）栈和队列

栈的定义

栈是仅在表尾进行插入、删除操作的线性表。表尾称为栈顶（Top），表头称为栈底（Base）。

插入元素到栈顶的操作，称为入栈。

从栈顶删除元素的操作，称为出栈。

栈的特点

后进先出（LIFO）

栈的抽象数据类型

ADT Stack {
    数据对象：D = {aᵢ | aᵢ ∈ ElemSet, i = 1,2,...,n, n ≥ 0}
    数据关系：R = {<aᵢ₋₁, aᵢ> | aᵢ₋₁, aᵢ ∈ D, i = 2,...,n}
    基本操作：
        // 1. 初始化
        InitStack(&S)
        // 2. 销毁
        DestroyStack(&S)
        // 3. 清空
        ClearStack(&S)
        // 4. 判空
        StackEmpty(S)
        // 5. 求表长
        StackLength(S)
        // 6. 取栈顶元素
        GetTop(S)
        // 7. 入栈
        Push(&S, e)
        // 8. 出栈
        Pop(&S, &e)
        // 9. 遍历
        StackTraverse(S)
} ADT Stack

栈的顺序存储结构

定义

typedef struct {
    // 1. 栈底指针，指向动态分配的数组首地址
    ElemType *base;
    // 2. 栈顶指针，指向栈顶元素的下一个位置
    ElemType *top;
    // 3. 栈的最大容量
    int stacksize;
} SeqStack;

初始化

// InitStack 初始化顺序栈
Status InitStack(SeqStack &S) {
    // 1. 新建一个 ElemType 类型的一维数组，并为该数组申请一块指定大小的内存空间，让栈底指针 base 指向该数组
    S.base = new ElemType[MAXSIZE];
    // 2. 判断栈底指针是否为空，为空则空间分配失败，直接退出
    if (!S.base) exit(OVERFLOW);
    // 3. 将栈顶指针 top 初始化为栈底指针，表示空栈
    S.top = S.base;
    // 4. 将栈的最大容量设置为 MAXSIZE
    S.stacksize = MAXSIZE;
    return OK;
}

销毁

// DestroyStack 销毁顺序栈
Status DestroyStack(SeqStack &S) {
    // 1. 判断栈底指针是否为空，为空则没有需要销毁的空间，直接返回
    if (S.base) {
        // 2. 释放栈底指针指向的数组占用的内存空间
        delete[] S.base;
        // 3. 将栈的最大容量置为 0
        S.stacksize = 0;
        // 4. 将栈底指针和栈顶指针都置为 NULL
        S.base = S.top = NULL;
    }
    return OK;
}

清空

// ClearStack 清空顺序栈
Status ClearStack(SeqStack &S) {
    // 1. 判断栈底指针是否为空，为空则无需清空
    if (S.base) 
        // 2. 将栈顶指针重置为栈底指针，表示空栈
        S.top = S.base;
    return OK;
}

判空

// StackEmpty 判断顺序栈是否为空
Status StackEmpty(const SeqStack &S) {
    // 1. 判断栈顶指针是否等于栈底指针，是则返回 TRUE，否则返回 FALSE
    if (S.top == S.base)
        return TRUE;
    else
        return FALSE;
}

求表长

// StackLength 返回顺序栈的长度
int StackLength(const SeqStack &S) {
    // 1. 直接返回栈顶指针与栈底指针的差值，即为栈中元素个数
    return S.top - S.base;
}

取栈顶元素

// GetTop 获取顺序栈的栈顶元素，并用 e 返回其值
Status GetTop(const SeqStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 ERROR
    if (S.top == S.base) return ERROR;
    // 2. 将栈顶元素（栈顶指针前一个位置的元素）的值赋给 e
    e = *(S.top - 1);
    return OK;
}

入栈

// Push 向顺序栈中插入元素 e 作为新的栈顶元素
Status Push(SeqStack &S, ElemType e) {
    // 1. 判断栈是否已满，若满则返回 ERROR
    if (S.top - S.base == S.stacksize) return ERROR;
    // 2. 将元素 e 放入栈顶指针指向的位置
    *S.top = e;
    // 3. 栈顶指针上移一位
    S.top++;
    return OK;
}

出栈

// Pop 删除顺序栈的栈顶元素，并用 e 返回其值
Status Pop(SeqStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 ERROR
    if (S.top == S.base) return ERROR;
    // 2. 栈顶指针下移一位，指向栈顶元素
    S.top--;
    // 3. 将栈顶元素的值赋给 e
    e = *S.top;
    return OK;
}

遍历

// StackTraverse 遍历顺序栈（从栈底到栈顶依次访问每个元素）
void StackTraverse(const SeqStack &S) {
    // 1. 定义临时指针，从栈底开始遍历
    ElemType *p = S.base;
    // 2. 循环遍历，直到临时指针到达栈顶位置
    while (p != S.top) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", *p);
        // 4. 指针后移，访问下一个元素
        p++;
    }
    // 换行，格式化输出
    printf("\n");
}

// StackTraverse 遍历顺序栈（自定义访问函数，更通用）
void StackTraverse(const SeqStack &S, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从栈底开始遍历
    ElemType *p = S.base;
    // 2. 循环遍历所有栈元素
    while (p != S.top) {
        // 3. 调用自定义函数访问元素
        visit(*p);
        // 4. 指针后移
        p++;
    }
}

栈的链式存储结构

定义

typedef struct StackNode{
    // 1. 结点数据域，存储栈元素
    ElemType data;
    // 2. 结点指针域，指向后继结点
    struct StackNode *next;
} StackNode, *LinkStack;

初始化

// InitStack 初始化链栈
Status InitStack(LinkStack &S) {
    // 1. 将栈顶指针置为 NULL，表示空栈
    S.top = NULL;
    return OK;
}

销毁

// DestroyStack 销毁链栈
Status DestroyStack(LinkStack &S) {
    // 1. 定义临时指针，用于遍历释放结点
    StackNode *p;
    // 2. 循环释放所有结点，直到栈顶指针为 NULL
    while (S.top) {
        // 3. 临时指针指向当前栈顶结点
        p = S.top;
        // 4. 栈顶指针下移，指向下一个结点
        S.top = S.top->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
    return OK;
}

清空

// ClearStack 清空链栈
Status ClearStack(LinkStack &S) {
    // 1. 链栈的清空操作与销毁操作逻辑一致，释放所有结点并将栈顶指针置为 NULL
    // 定义临时指针，用于遍历释放结点
    StackNode *p;
    // 2. 循环释放所有结点，直到栈顶指针为 NULL
    while (S.top) {
        // 3. 临时指针指向当前栈顶结点
        p = S.top;
        // 4. 栈顶指针下移，指向下一个结点
        S.top = S.top->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
    return OK;
}

判空

// StackEmpty 判断链栈是否为空
Status StackEmpty(const LinkStack &S) {
    // 1. 判断栈顶指针是否为 NULL，是则返回 TRUE，否则返回 FALSE
    if (S.top == NULL)
        return TRUE;
    else
        return FALSE;
}

求表长

// StackLength 返回链栈的长度
int StackLength(const LinkStack &S) {
    // 1. 定义计数器，初始化为 0
    int len = 0;
    // 2. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 3. 循环遍历所有结点
    while (p) {
        // 4. 计数器加 1
        len++;
        // 5. 指针后移，访问下一个结点
        p = p->next;
    }
    // 6. 返回栈的长度
    return len;
}

取栈顶元素

// GetTop 获取链栈的栈顶元素，并用 e 返回其值
Status GetTop(const LinkStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 ERROR
    if (S.top == NULL) return ERROR;
    // 2. 将栈顶结点的数据域的值赋给 e
    e = S.top->data;
    return OK;
}

入栈

// Push 向链栈中插入元素 e 作为新的栈顶元素
Status Push(LinkStack &S, ElemType e) {
    // 1. 新建一个结点，为其分配内存空间
    StackNode *p = new StackNode;
    // 2. 判断新结点是否分配成功，失败则退出
    if (!p) exit(OVERFLOW);
    // 3. 将元素 e 赋给新结点的数据域
    p->data = e;
    // 4. 将新结点的指针域指向当前的栈顶结点
    p->next = S.top;
    // 5. 将栈顶指针更新为新结点
    S.top = p;
    return OK;
}

出栈

// Pop 删除链栈的栈顶元素，并用 e 返回其值
Status Pop(LinkStack &S, ElemType &e) {
    // 1. 判断栈是否为空，若空则返回 ERROR
    if (S.top == NULL) return ERROR;
    // 2. 定义临时指针，指向当前栈顶结点
    StackNode *p = S.top;
    // 3. 将栈顶结点的数据域的值赋给 e
    e = p->data;
    // 4. 将栈顶指针下移，指向下一个结点
    S.top = S.top->next;
    // 5. 释放临时指针指向的结点
    delete p;
    return OK;
}

遍历

// StackTraverse 遍历链栈（从栈顶到栈底依次访问每个元素）
void StackTraverse(const LinkStack &S) {
    // 1. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 2. 循环遍历所有结点
    while (p) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", p->data);
        // 4. 指针后移，访问下一个结点
        p = p->next;
    }
    // 换行，格式化输出
    printf("\n");
}

// StackTraverse 遍历链栈（自定义访问函数，更通用）
void StackTraverse(const LinkStack &S, void (*visit)(ElemType)) {
    // 1. 定义临时指针，从栈顶开始遍历
    StackNode *p = S.top;
    // 2. 循环遍历所有结点
    while (p) {
        // 3. 调用自定义函数访问元素
        visit(p->data);
        // 4. 指针后移
        p = p->next;
    }
}

队列的定义

队列是仅在表尾插入元素，仅在在表头删除元素的线性表。

插入元素到表尾的操作，称为入队。

从表头删除元素的操作，称为出队。

队列的特点

先进先出（FIFO）

队列的抽象数据类型

ADT Queue {
    数据对象：D = {aᵢ | aᵢ ∈ ElemSet, i = 1,2,...,n, n ≥ 0}
    数据关系：R = {<aᵢ₋₁, aᵢ> | aᵢ₋₁, aᵢ ∈ D, i = 2,...,n}
    基本操作：
        // 1. 初始化
        InitQueue(&Q)
        // 2. 销毁
        DestroyQueue(&Q)
        // 3. 清空
        ClearQueue(&Q)
        // 4. 判空
        QueueEmpty(Q)
        // 5. 求表长
        QueueLength(Q)
        // 6. 取队头元素
        GetHead(Q)
        // 7. 入队
        EnQueue(&Q, e)
        // 8. 出队
        DeQueue(&Q, &e)
        // 9. 遍历
        QueueTraverse(Q)
} ADT Queue

队列的顺序存储结构

定义

typedef struct {
    // 1. 动态分配的数组首地址，存储队列所有元素
    QElemType *base;
    // 2. 队头指针，指向队列的头元素
    int front;
    // 3. 队尾指针，指向队列尾元素的下一个位置
    int rear;
} SqQueue;

初始化

// InitQueue 初始化顺序循环队列
Status InitQueue(SqQueue &Q) {
    // 1. 新建一个 QElemType 类型的一维数组，申请指定大小的内存空间
    Q.base = new QElemType[MAXQSIZE];
    // 2. 判断内存分配是否失败，失败则直接退出程序
    if (!Q.base) exit(OVERFLOW);
    // 3. 队头和队尾指针都置为 0，表示空队列
    Q.front = Q.rear = 0;
    return OK;
}

销毁

// DestroyQueue 销毁顺序循环队列
Status DestroyQueue(SqQueue &Q) {
    // 1. 判断数组指针是否为空，不为空则释放其占用的内存
    if (Q.base) {
        delete[] Q.base;
        // 2. 将数组指针置为 NULL
        Q.base = NULL;
    }
    // 3. 队头和队尾指针都置为 0
    Q.front = Q.rear = 0;
    return OK;
}

清空

// ClearQueue 清空顺序循环队列
Status ClearQueue(SqQueue &Q) {
    // 1. 直接将队头和队尾指针都置为 0，即可清空队列
    Q.front = Q.rear = 0;
    return OK;
}

判空

// QueueEmpty 判断顺序循环队列是否为空
Status QueueEmpty(const SqQueue &Q) {
    // 1. 判断队头指针是否等于队尾指针，是则返回 TRUE，否则返回 FALSE
    if (Q.front == Q.rear)
        return TRUE;
    else
        return FALSE;
}

求表长

// QueueLength 返回顺序循环队列的长度
int QueueLength(const SqQueue &Q) {
    // 1. 循环队列长度计算公式：(队尾指针 - 队头指针 + 最大容量) % 最大容量
    return (Q.rear - Q.front + MAXQSIZE) % MAXQSIZE;
}

取队头元素

// GetHead 获取顺序循环队列的队头元素，并用 e 返回其值
Status GetHead(const SqQueue &Q, QElemType &e) {
    // 1. 判断队列是否为空，为空则返回 ERROR
    if (Q.front == Q.rear) return ERROR;
    // 2. 将队头元素的值赋给 e
    e = Q.base[Q.front];
    return OK;
}

入队

// EnQueue 向顺序循环队列中插入元素 e 作为新的队尾元素
Status EnQueue(SqQueue &Q, QElemType e) {
    // 1. 判断队列是否已满，满则返回 ERROR
    if ((Q.rear + 1) % MAXQSIZE == Q.front) return ERROR;
    // 2. 将元素 e 放入队尾指针指向的位置
    Q.base[Q.rear] = e;
    // 3. 队尾指针循环后移一位
    Q.rear = (Q.rear + 1) % MAXQSIZE;
    return OK;
}

出队

// DeQueue 删除顺序循环队列的队头元素，并用 e 返回其值
Status DeQueue(SqQueue &Q, QElemType &e) {
    // 1. 判断队列是否为空，为空则返回 ERROR
    if (Q.front == Q.rear) return ERROR;
    // 2. 将队头元素的值赋给 e
    e = Q.base[Q.front];
    // 3. 队头指针循环后移一位
    Q.front = (Q.front + 1) % MAXQSIZE;
    return OK;
}

遍历

// QueueTraverse 遍历顺序循环队列（从队头到队尾依次访问每个元素）
void QueueTraverse(const SqQueue &Q) {
    // 1. 定义临时指针，从队头开始遍历
    int p = Q.front;
    // 2. 循环遍历，直到临时指针到达队尾位置
    while (p != Q.rear) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", Q.base[p]);
        // 4. 临时指针循环后移一位
        p = (p + 1) % MAXQSIZE;
    }
    // 换行，格式化输出
    printf("\n");
}

// QueueTraverse 遍历顺序循环队列（自定义访问函数，更通用）
void QueueTraverse(const SqQueue &Q, void (*visit)(QElemType)) {
    // 1. 定义临时指针，从队头开始遍历
    int p = Q.front;
    // 2. 循环遍历所有队列元素
    while (p != Q.rear) {
        // 3. 调用自定义函数访问元素
        visit(Q.base[p]);
        // 4. 临时指针循环后移一位
        p = (p + 1) % MAXQSIZE;
    }
}

队列的链式存储结构

定义

// 链队结点结构
typedef struct QNode {
    // 1. 结点数据域，存储队列元素
    QElemType data;
    // 2. 结点指针域，指向后继结点
    struct QNode *next;
} QNode;

// 链队结构
typedef struct {
    // 1. 队头指针，指向头结点
    QNode *front;
    // 2. 队尾指针，指向队尾结点
    QNode *rear;
} LinkQueue;

初始化

// InitQueue 初始化链队（带头结点）
Status InitQueue(LinkQueue &Q) {
    // 1. 新建一个头结点，为其分配内存空间
    Q.front = Q.rear = new QNode;
    // 2. 判断头结点是否分配成功，失败则退出
    if (!Q.front) exit(OVERFLOW);
    // 3. 将头结点的指针域置为 NULL，表示空队列
    Q.front->next = NULL;
    return OK;
}

销毁

// DestroyQueue 销毁链队
Status DestroyQueue(LinkQueue &Q) {
    // 1. 定义临时指针，用于遍历释放结点
    QNode *p;
    // 2. 循环释放所有结点，包括头结点
    while (Q.front) {
        // 3. 临时指针指向当前头结点
        p = Q.front;
        // 4. 队头指针后移，指向下一个结点
        Q.front = Q.front->next;
        // 5. 释放临时指针指向的结点
        delete p;
    }
    // 6. 释放后头结点指针已为 NULL，队尾指针也置为 NULL
    Q.rear = NULL;
    return OK;
}

清空

// ClearQueue 清空链队（保留头结点）
Status ClearQueue(LinkQueue &Q) {
    // 1. 定义临时指针，指向队头结点的下一个结点（第一个数据结点）
    QNode *p = Q.front->next;
    // 2. 定义临时指针，用于释放结点
    QNode *q;
    // 3. 循环释放所有数据结点
    while (p) {
        // 4. 临时指针 q 指向当前要释放的结点
        q = p;
        // 5. 指针 p 后移，指向下一个结点
        p = p->next;
        // 6. 释放临时指针 q 指向的结点
        delete q;
    }
    // 7. 队头结点的指针域置为 NULL
    Q.front->next = NULL;
    // 8. 队尾指针重新指向头结点
    Q.rear = Q.front;
    return OK;
}

判空

// QueueEmpty 判断链队是否为空
Status QueueEmpty(const LinkQueue &Q) {
    // 1. 判断队头指针的下一个结点是否为 NULL，是则返回 TRUE，否则返回 FALSE
    if (Q.front->next == NULL)
        return TRUE;
    else
        return FALSE;
}

求表长

// QueueLength 返回链队的长度
int QueueLength(const LinkQueue &Q) {
    // 1. 定义计数器，初始化为 0
    int len = 0;
    // 2. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 3. 循环遍历所有数据结点
    while (p) {
        // 4. 计数器加 1
        len++;
        // 5. 指针后移，访问下一个结点
        p = p->next;
    }
    // 6. 返回队列的长度
    return len;
}

取队头元素

// GetHead 获取链队的队头元素，并用 e 返回其值
Status GetHead(const LinkQueue &Q, QElemType &e) {
    // 1. 判断队列是否为空，为空则返回 ERROR
    if (Q.front->next == NULL) return ERROR;
    // 2. 将第一个数据结点的数据域的值赋给 e
    e = Q.front->next->data;
    return OK;
}

入队

// EnQueue 向链队中插入元素 e 作为新的队尾元素
Status EnQueue(LinkQueue &Q, QElemType e) {
    // 1. 新建一个结点，为其分配内存空间
    QNode *p = new QNode;
    // 2. 判断新结点是否分配成功，失败则退出
    if (!p) exit(OVERFLOW);
    // 3. 将元素 e 赋给新结点的数据域
    p->data = e;
    // 4. 将新结点的指针域置为 NULL
    p->next = NULL;
    // 5. 将新结点插入到队尾结点的后面
    Q.rear->next = p;
    // 6. 将队尾指针更新为新结点
    Q.rear = p;
    return OK;
}

出队

// DeQueue 删除链队的队头元素，并用 e 返回其值
Status DeQueue(LinkQueue &Q, QElemType &e) {
    // 1. 判断队列是否为空，为空则返回 ERROR
    if (Q.front->next == NULL) return ERROR;
    // 2. 定义临时指针，指向第一个数据结点（队头元素所在结点）
    QNode *p = Q.front->next;
    // 3. 将队头元素的数据域的值赋给 e
    e = p->data;
    // 4. 将头结点的指针域指向第二个数据结点
    Q.front->next = p->next;
    // 5. 判断删除的是否为队尾结点，若是则将队尾指针重新指向头结点
    if (Q.rear == p)
        Q.rear = Q.front;
    // 6. 释放临时指针指向的结点
    delete p;
    return OK;
}

遍历

// QueueTraverse 遍历链队（从队头到队尾依次访问每个元素）
void QueueTraverse(const LinkQueue &Q) {
    // 1. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 2. 循环遍历所有数据结点
    while (p) {
        // 3. 访问当前元素（这里以打印为例）
        printf("%c ", p->data);
        // 4. 指针后移，访问下一个结点
        p = p->next;
    }
    // 换行，格式化输出
    printf("\n");
}

// QueueTraverse 遍历链队（自定义访问函数，更通用）
void QueueTraverse(const LinkQueue &Q, void (*visit)(QElemType)) {
    // 1. 定义临时指针，从第一个数据结点开始遍历
    QNode *p = Q.front->next;
    // 2. 循环遍历所有数据结点
    while (p) {
        // 3. 调用自定义函数访问元素
        visit(p->data);
        // 4. 指针后移
        p = p->next;
    }
}
